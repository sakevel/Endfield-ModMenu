cmake_minimum_required(VERSION 3.25)
project(EndfieldModMenu VERSION 0.2.1 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 is required")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_options(/utf-8 /W4 /permissive-)
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
add_library(ModMenu SHARED src/mod.cpp)
target_include_directories(ModMenu PRIVATE sdk/include src)
set_target_properties(ModMenu PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/package/$<CONFIG>/mod-menu")
add_custom_target(package ALL DEPENDS ModMenu)
foreach(name mod.ini mod-menu.lua model.lua native-ui.lua config.ini icon.png zml-package.json)
  add_custom_command(TARGET package POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy
    "${CMAKE_CURRENT_SOURCE_DIR}/mod/${name}" "${CMAKE_BINARY_DIR}/package/$<CONFIG>/mod-menu/${name}")
endforeach()
enable_testing()
add_executable(PatchTests tests/patch_tests.cpp)
target_include_directories(PatchTests PRIVATE src)
add_test(NAME patch COMMAND PatchTests)
set(ZML_LUPA_PATH "" CACHE PATH "Optional existing Lupa installation")
set(ZML_SERVICES_TEST_EXE "" CACHE FILEPATH "Optional loader black-box ServicesTests executable")
if(ZML_LUPA_PATH AND ZML_SERVICES_TEST_EXE)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME lua_behavior COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/lua_behavior_tests.py"
    --lupa-dir "${ZML_LUPA_PATH}" --services "${ZML_SERVICES_TEST_EXE}")
  set_tests_properties(lua_behavior PROPERTIES TIMEOUT 30)
endif()
